def get_fechas_edp():
    hoy = datetime.now().date()
    if hoy.day <= 20:
        fecha_fin = date(hoy.year, hoy.month, 20)
        inicio_mes = date(hoy.year, hoy.month, 1)
        fecha_inicio = (inicio_mes - timedelta(days=1)).replace(day=21)
    else:
        fecha_inicio = date(hoy.year, hoy.month, 21)
        prox_mes = hoy.month + 1 if hoy.month < 12 else 1
        prox_anio = hoy.year if hoy.month < 12 else hoy.year + 1
        fecha_fin = date(prox_anio, prox_mes, 20)
    return pd.Timestamp(fecha_inicio), pd.Timestamp(fecha_fin)

def get_kpi_bono_inteligente_v2():
    """
    Cálculo con Lógica de 'Descuento de Especiales':
    Como el operador pesa todo junto (Mix KG + UN), hacemos ingeniería inversa:
    Kg Reales Ducto = (Peso Total Balanza) - (Peso Teórico de Piezas Especiales).
    """
    if not os.path.exists(DB_PROD): return 0, 0, date.today(), date.today()
    
    start_date, end_date = get_fechas_edp()
    conn = sqlite3.connect(DB_PROD)
    
    try:
        # 1. Traer Datos
        df_ped = pd.read_sql("SELECT * FROM pedidos WHERE estado='Terminado'", conn)
        df_items = pd.read_sql("SELECT * FROM items_pedido", conn)
        conn.close()
        
        if df_ped.empty or df_items.empty: return 0, 0, start_date.date(), end_date.date()

        # 2. Filtro Fechas
        df_ped['fecha_termino'] = pd.to_datetime(df_ped['fecha_termino'], errors='coerce')
        mask = (df_ped['fecha_termino'] >= start_date) & (df_ped['fecha_termino'] <= end_date)
        df_ped = df_ped[mask]
        
        if df_ped.empty: return 0, 0, start_date.date(), end_date.date()
        
        # 3. Merge
        df_full = pd.merge(df_items, df_ped, left_on='pedido_id', right_on='id', suffixes=('_item', '_ped'))
        
        total_kg_bono = 0.0
        total_plata_edp = 0.0
        
        # 4. Cálculo Pedido a Pedido
        for pid in df_full['pedido_id'].unique():
            df_p = df_full[df_full['pedido_id'] == pid].copy()
            
            # Peso reportado por operador (Mix Total)
            peso_balanza_total = df_p.iloc[0]['kg_reales'] if pd.notna(df_p.iloc[0]['kg_reales']) else 0.0
            
            # --- SEPARACIÓN DE AGUAS ---
            # Identificar qué es Especial (UN) y qué es Normal (KG)
            mask_un = df_p['unidad_cobro'].str.lower().str.strip() == 'un'
            mask_kg = ~mask_un
            
            # Peso Teórico de las Especiales (Asumimos que es correcto porque se cobran por unidad)
            peso_teorico_especiales = df_p.loc[mask_un, 'peso_total'].sum()
            
            # Peso Teórico de los Normales
            peso_teorico_normales = df_p.loc[mask_kg, 'peso_total'].sum()
            
            # --- INGENIERÍA INVERSA ---
            # Calculamos cuánto pesa REALMENTE la parte de ductos normales
            # Si el operador pesó 0 o menos que las especiales, asumimos el teórico (Plan B)
            if peso_balanza_total <= peso_teorico_especiales:
                peso_real_normales_calculado = peso_teorico_normales
            else:
                peso_real_normales_calculado = peso_balanza_total - peso_teorico_especiales
            
            # --- CÁLCULO DE DINERO ---
            monto_pedido = 0.0
            
            # A) Plata de Especiales (Precio Fijo x Cantidad) - No le afecta el peso
            monto_pedido += df_p.loc[mask_un, 'total_linea'].sum()
            
            # B) Plata de Normales (Ajustada por el peso real calculado)
            plata_teorica_normales = df_p.loc[mask_kg, 'total_linea'].sum()
            
            factor = 1.0
            if peso_teorico_normales > 0:
                factor = peso_real_normales_calculado / peso_teorico_normales
            
            monto_pedido += (plata_teorica_normales * factor)
            
            # --- ACUMULADORES FINALES ---
            # Para el BONO: Sumamos el peso real calculado de normales + peso teórico de especiales
            # (Al final, el bono es por Kg procesados totales)
            total_kg_bono += (peso_real_normales_calculado + peso_teorico_especiales)
            
            total_plata_edp += monto_pedido
            
        return total_kg_bono, total_plata_edp, start_date.date(), end_date.date()

    except Exception as e:
        print(f"Error KPI: {e}")
        return 0, 0, start_date.date(), end_date.date()
