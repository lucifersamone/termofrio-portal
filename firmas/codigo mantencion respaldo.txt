def init_db_mant():
    conn = sqlite3.connect(DB_PATH)
    c = conn.cursor()
    c.execute('''CREATE TABLE IF NOT EXISTS registros_inspeccion (
        fecha_hora TEXT, maquina_id TEXT, operador TEXT, checks_ok TEXT, 
        repuesto_necesario INTEGER, comentarios TEXT, firma_path TEXT, pdf_path TEXT
    )''')
    try: c.execute("ALTER TABLE registros_inspeccion ADD COLUMN pdf_path TEXT")
    except: pass
    
    c.execute('''CREATE TABLE IF NOT EXISTS maquinas (
        id TEXT PRIMARY KEY, nombre TEXT, caracteristicas TEXT, frecuencia_mantencion TEXT, foto_path TEXT, modelo TEXT
    )''')
    try: c.execute("ALTER TABLE maquinas ADD COLUMN modelo TEXT")
    except: pass
    
    c.execute('''CREATE TABLE IF NOT EXISTS plan_externo (
        id INTEGER PRIMARY KEY AUTOINCREMENT, maquina_id TEXT, fecha_programada DATE, 
        proveedor TEXT, estado TEXT, evidencia_pdf_path TEXT, fecha_realizacion DATE,
        evaluacion_proveedor TEXT
    )''')
    try: c.execute("ALTER TABLE plan_externo ADD COLUMN evaluacion_proveedor TEXT")
    except: pass
    conn.commit(); conn.close()

def obtener_puntos(maquina_id):
    m = maquina_id.upper()
    
    # GRUPO C: MAQUINAS AVANZADAS
    if "CNC" in m or "PLASMA" in m or "COILINE" in m:
        puntos = ["Limpieza General", "Tablero Eléctrico y Cables", "PC y Pantalla", "Sist. Aire / Hidráulico", "Rieles/Guías", "Consumibles/Corte"]
        if "COILINE" in m: puntos.append("Carretes y Rodamientos")
        return puntos
        
    # GRUPO A: MAQUINAS MANUALES (Excluyendo eléctricas explícitas)
    if ("RODONADORA" in m and "ELECTRICA" not in m) or "CILINDRADORA" in m:
        return ["Limpieza General", "Engrase / Lubricación", "Estado de Rodillos/Cuchillos", "Ajuste Manivelas"]
        
    # GRUPO B: MAQUINAS ELÉCTRICAS (Resto)
    return ["Limpieza General", "Tablero Eléctrico y Cables", "Funcionamiento Motor", "Rodamientos y Guías", "Protecciones Seguridad"]

def ya_realizado_hoy(maquina_id):
    conn = sqlite3.connect(DB_PATH)
    hoy = datetime.now().strftime('%Y-%m-%d')
    cursor = conn.cursor()
    cursor.execute("SELECT operador, fecha_hora FROM registros_inspeccion WHERE maquina_id=? AND fecha_hora LIKE ?", (maquina_id, f"{hoy}%"))
    dato = cursor.fetchone()
    conn.close()
    return dato

def generar_pdf_checklist(maquina, operador, fecha_dt, checks, estado, obs, firma_path):
    """Genera PDF con diseño profesional, colores y validación visual."""
    nombre_seguro = "".join([c if c.isalnum() else "_" for c in maquina])
    fecha_str_file = fecha_dt.strftime('%Y%m%d_%H%M%S')
    nombre_pdf = f"{nombre_seguro}_{fecha_str_file}.pdf"
    ruta_pdf = os.path.join(CARPETA_INFORMES, nombre_pdf)
    
    try:
        c = canvas.Canvas(ruta_pdf, pagesize=letter)
        w, h = letter
        
        # --- ENCABEZADO CON COLOR ---
        c.setFillColor(colors.darkblue) # Barra Azul
        c.rect(0, h - 80, w, 80, fill=1, stroke=0)
        
        c.setFillColor(colors.white) # Texto Blanco
        c.setFont("Helvetica-Bold", 18)
        c.drawCentredString(w/2, h - 50, "REPORTE DE INSPECCIÓN - TERMOFRIO SPA")
        
        # --- INFORMACIÓN GENERAL (Centrada) ---
        c.setFillColor(colors.black)
        y_info = h - 110
        c.setFont("Helvetica-Bold", 12)
        c.drawString(50, y_info, f"Máquina:")
        c.setFont("Helvetica", 12)
        c.drawString(120, y_info, maquina)
        
        c.setFont("Helvetica-Bold", 12)
        c.drawString(350, y_info, f"Fecha:")
        c.setFont("Helvetica", 12)
        c.drawString(400, y_info, fecha_dt.strftime('%d/%m/%Y %H:%M'))
        
        y_info -= 20
        c.setFont("Helvetica-Bold", 12)
        c.drawString(50, y_info, f"Operador:")
        c.setFont("Helvetica", 12)
        c.drawString(120, y_info, operador)
        
        # --- ESTADO GRANDE Y COLOREADO ---
        y_info -= 40
        c.setFont("Helvetica-Bold", 16)
        c.drawCentredString(w/2, y_info, "ESTADO FINAL DEL EQUIPO")
        
        y_info -= 25
        c.setFont("Helvetica-Bold", 20)
        if estado == "CON FALLA":
            c.setFillColor(colors.red)
            texto_estado = "⛔ NO OPERATIVO / CON FALLA"
        else:
            c.setFillColor(colors.green)
            texto_estado = "✅ OPERATIVO"
        
        c.drawCentredString(w/2, y_info, texto_estado)
        c.setFillColor(colors.black) # Volver a negro
        
        # Línea divisoria
        c.setStrokeColor(colors.gray)
        c.line(50, y_info - 20, w - 50, y_info - 20)
        
        # --- CHECKLIST (Puntos en Verde) ---
        y = y_info - 50
        c.setFont("Helvetica-Bold", 12)
        c.drawString(50, y, "Puntos de Control Verificados:")
        y -= 25
        c.setFont("Helvetica", 11)
        
        for item in checks:
            # Dibujar el [OK] en verde
            c.setFillColor(colors.green)
            c.drawString(60, y, "[OK]")
            
            # Dibujar el texto en negro
            c.setFillColor(colors.black)
            c.drawString(100, y, item)
            
            y -= 20
            if y < 100: c.showPage(); y = h - 50
            
        # --- OBSERVACIONES ---
        y -= 20
        c.setFillColor(colors.darkblue)
        c.rect(50, y, w-100, 20, fill=1, stroke=0)
        c.setFillColor(colors.white)
        c.setFont("Helvetica-Bold", 12)
        c.drawString(60, y+5, "Observaciones Registradas")
        
        y -= 20
        c.setFillColor(colors.black)
        c.setFont("Helvetica", 10)
        if obs.strip():
            text_object = c.beginText(60, y)
            text_object.textLines(obs)
            c.drawText(text_object)
        else:
            c.drawString(60, y, "Sin observaciones adicionales.")
            
        # --- FIRMA ---
        y_firma = 100
        c.setStrokeColor(colors.black)
        c.line(w/2 - 100, y_firma, w/2 + 100, y_firma) # Línea de firma
        
        c.setFont("Helvetica", 10)
        c.drawCentredString(w/2, y_firma - 15, f"Firmado digitalmente por: {operador}")
        
        if os.path.exists(firma_path):
            try:
                # Ajustar imagen centrada sobre la línea
                c.drawImage(ImageReader(firma_path), w/2 - 60, y_firma + 5, width=120, height=60, mask='auto')
            except:
                c.drawString(w/2, y_firma + 20, "[Error cargando imagen firma]")
            
        c.save()
        return ruta_pdf
    except Exception as e:
        print(f"Error generando PDF: {e}")
        return None

def calcular_estado_mes(fecha_prog, estado_db):
    if estado_db == 'Realizado': return "✅ OK"
    hoy = datetime.now().date()
    try: prog = pd.to_datetime(fecha_prog).date()
    except: return "⚪ Error"
    if hoy > prog and (hoy.month > prog.month or hoy.year > prog.year): return "🔴 Vencido"
    if hoy.month == prog.month and hoy.year == prog.year:
        ultimo_dia = (prog.replace(day=1) + timedelta(days=32)).replace(day=1) - timedelta(days=1)
        try: dias_habiles = np.busday_count(hoy, ultimo_dia)
        except: dias_habiles = 30
        return "🟠 ALERTA (2 Sem)" if dias_habiles <= 10 else "🔵 En Plazo"
    return "📅 Prog."
